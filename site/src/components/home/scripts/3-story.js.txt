  if (m) { const s = secs[m[1] - 1]; document.body.classList.add('review'); s.sec.classList.add('review-on'); s.cur = s.tgt = m[2] / 100; render(s); return; }

  // v32: readout rows follow the agent's steps; the step bar follows the sections
  secs.forEach(s => { s.ro = [...s.sec.querySelectorAll('.ro')]; });
  const _render = render;
  render = function (s) {
    _render(s);
    let li = 0; s.at.forEach((a, i) => { if (s.cur >= a) li = i; });
    const fin = s.cur >= s.done;
    if (s.sec.id === 'search') searchMap.draw(s.cur);
    if (s.sec.id === 'score' && window.ScoreFlow) { s.flow = s.flow || ScoreFlow(s.sec.querySelector('.scr')); s.flow(s.cur); }
    s.sec.classList.toggle('live', s.cur > 0.004 || small.matches || reduce);
    s.ro.forEach((r, i) => { r.classList.toggle('done', fin || i < li); r.classList.toggle('on', !fin && i === li); });
    bar();
  };
  const barLinks = [...document.querySelectorAll('.agentbar a')];
  function bar() {
    const top = navH() + 64; let act = -1;
    secs.forEach((s, i) => { const r = s.sec.getBoundingClientRect(); if (r.top <= top && r.bottom > top) act = i; });
    barLinks.forEach((a, i) => { const s = secs[i]; a.style.setProperty('--f', s.cur.toFixed(3)); a.classList.toggle('on', i === act); a.classList.toggle('done', s.cur >= s.done && i !== act); });
  }
  // fit each 640×440 drawing to its stage: as large as the width and viewport allow
  function fit() {
    document.querySelectorAll('.v32 .v-stagebox').forEach(box => {
      const st = box.querySelector('.stage');
      if (small.matches) { st.style.zoom = ''; return; }
      const head = box.closest('.sgrid'); const cap = head.querySelector('.agent-panel'); const vh = head.querySelector('.v-head');
      const availH = innerHeight - navH() - 60 - vh.offsetHeight - (head.querySelector('.v-body .funnel') ? head.querySelector('.v-body .funnel').offsetHeight + 40 : 0) - 24;
      const z = Math.max(.6, Math.min((box.clientWidth) / 640, availH / 440, 1.25));
      st.style.zoom = z.toFixed(3);
      { const room = innerHeight - navH() - 60, over = head.scrollHeight - room; if (over > 0) st.style.zoom = Math.max(.6, z - over / 440).toFixed(3); }
      if (st.classList.contains('srch')) searchMap.resize(z);
    });
  }
  addEventListener('resize', fit); fit();

  addEventListener('scroll', () => { kick(); bar(); }, { passive: true }); addEventListener('resize', kick);
  measure(); secs.forEach(s => { s.cur = s.tgt; render(s); });
})();
