
(() => {
  // who-it's-for cards: the query shows first, then its results settle in
  const cards = [...document.querySelectorAll('.jobs .job')];
  cards.forEach(card => {
    const items = [...card.querySelectorAll('.rank-preview > div, .jviz circle, .jviz text')];
    const step = items.length > 6 ? 14 : 140;
    items.forEach((el, i) => { el.classList.add('rv'); el.style.setProperty('--d', (380 + i * step) + 'ms'); });
  });
  const io = new IntersectionObserver(es => es.forEach(e => { if (e.isIntersecting) { e.target.classList.add('in'); io.unobserve(e.target); } }), { threshold: .45 });
  cards.forEach(c => io.observe(c));
})();
